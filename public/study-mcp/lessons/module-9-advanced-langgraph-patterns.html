Lesson 9.1

Reflection Agent

Generate → Critique → Revise, lặp tới khi đạt hoặc hết số lần cho phép.

Giống bạn viết luận, tự đọc lại, tự sửa — không ai giúp.
Lesson 9.2

Reflexion Agent

Actor dùng tool lấy bằng chứng thật trước khi Revisor phê bình — khác Reflection ở chỗ có evidence bên ngoài.

Giống tra Google trước khi tự phê bình báo cáo, không đoán mò.
Lesson 9.3

Corrective RAG (CRAG)

Chấm điểm độ liên quan tài liệu → nếu kém → fallback web search → mới generate.

Thủ thư: sách không đủ liên quan thì tự ra ngoài tra Google.
Lesson 9.4

Self-RAG & Adaptive RAG

Self-RAG: LLM tự hỏi "cần retrieve không? có hallucination không? có hữu ích không?". Adaptive RAG: router chọn 1 trong 3 chiến lược trước khi làm gì cả.

Lễ tân bệnh viện: phân loại triệu chứng rồi mới chuyển đúng khoa.

Code mẫu chạy được cho từng lesson (đã test, output thật)

Mỗi ví dụ dùng đúng StateGraph / add_conditional_edges / compile() / invoke() thật. Vì không có API key, bước "LLM" được thay bằng hàm rule-based — nhấn để mở/đóng.

🅰️ Lesson 9.1 — Reflection Agent (viết tweet, tự phê bình, tự sửa) 
from typing import TypedDict, List, Annotated
import operator
from langgraph.graph import StateGraph, START, END

class TweetState(TypedDict):
    topic: str
    tweet: str
    critique: str
    iteration: int
    log: Annotated[List[str], operator.add]

def generate(state: TweetState) -> dict:
    if state["iteration"] == 0:
        tweet = f"{state['topic']} moi ra mat, mua ngay!!!"
    else:
        tweet = f"{state['topic']} moi ra mat: nhe hon 20%, pin 2 ngay, gia khong doi. Dat truoc hom nay."
    return {"tweet": tweet, "log": [f"[GENERATE v{state['iteration']+1}] {tweet}"]}

def reflect(state: TweetState) -> dict:
    tweet = state["tweet"]
    if len(tweet) < 40 or "!!!" in tweet:
        critique = "Qua so sai, khong co thong tin cu the. Thieu thuyet phuc."
    else:
        critique = "DAT: co thong tin cu the, giong dieu chuyen nghiep."
    return {"critique": critique, "iteration": state["iteration"]+1,
            "log": [f"[REFLECT] {critique}"]}

def should_continue(state: TweetState) -> str:
    if "DAT" in state["critique"] or state["iteration"] >= 3:
        return "done"
    return "revise"

graph = StateGraph(TweetState)
graph.add_node("generate", generate)
graph.add_node("reflect", reflect)
graph.add_edge(START, "generate")
graph.add_edge("generate", "reflect")
graph.add_conditional_edges("reflect", should_continue, {"revise": "generate", "done": END})
app = graph.compile()

result = app.invoke({"topic": "Tai nghe XYZ", "tweet": "", "critique": "", "iteration": 0, "log": []})

🅱️ Lesson 9.2 — Reflexion Agent (Actor dùng tool tra cứu evidence) 
def actor_generate(state) -> dict:
    if state["iteration"] == 0:
        claim = "LangGraph duoc phat trien boi Google."   # sai, chua co evidence
    else:
        claim = "LangGraph la mot thu vien mo rong cua LangChain, phat trien boi LangChain Inc."
    return {"claim": claim, "log": [f"[ACTOR draft v{state['iteration']+1}] {claim}"]}

def tool_search(state) -> dict:  # Reflexion: TOOL that su duoc goi de lay bang chung
    fact_db = {"langgraph": "LangGraph la du an cua LangChain Inc, mo rong tu LangChain..."}
    return {"evidence": fact_db["langgraph"], "log": [f"[TOOL search] -> {fact_db['langgraph']}"]}

def revisor_critique(state) -> dict:  # phe binh DUA TREN evidence, khong doan mo
    if "google" in state["claim"].lower():
        critique = f"SAI: evidence cho thay '{state['evidence']}'. Hallucination, phai sua."
    elif "langchain inc" in state["claim"].lower():
        critique = "DAT: claim khop voi evidence tra duoc."
    else:
        critique = "Chua du chi tiet."
    return {"critique": critique, "iteration": state["iteration"]+1, "log": [f"[REVISOR] {critique}"]}

VECTORSTORE = {
    "checkpointer": ["MemorySaver luu state trong RAM...", "SqliteSaver luu state xuong file .db..."],
    "reducer": ["Cau chuyen ve Tam Quoc Dien Nghia thoi nha Han."],  # co tinh KHONG lien quan
}

def grade_documents(state) -> dict:  # grader: chi sinh nhan relevant/irrelevant
    def is_relevant(doc): return "tam quoc" not in doc.lower()
    relevant = [d for d in state["documents"] if is_relevant(d)]
    return {"relevant_docs": relevant, "web_search_needed": len(relevant) == 0}

def route_after_grade(state) -> str:
    return "web_search" if state["web_search_needed"] else "generate"

# graph: retrieve -> grade_documents -> [web_search neu thieu] -> generate
graph.add_conditional_edges("grade_documents", route_after_grade,
    {"web_search": "web_search", "generate": "generate"})
graph.add_edge("web_search", "generate")

🅳 Lesson 9.4 — Adaptive RAG (router chọn 1 trong 3 chiến lược) 
def classify_query(state) -> dict:  # router: chi 1 lan phan loai
    q = state["question"].lower()
    if any(w in q for w in ["hom nay", "moi nhat", "hien tai"]):
        strategy = "web_search"
    elif any(w in q for w in ["tai lieu", "checkpointer"]):
        strategy = "rag"
    else:
        strategy = "direct_answer"
    return {"strategy": strategy, "log": [f"[ROUTER] -> {strategy}"]}

# graph: classify -> (route theo strategy) -> direct_answer | rag | web_search -> END
graph.add_conditional_edges("classify", lambda s: s["strategy"],
    {"direct_answer": "direct_answer", "rag": "rag", "web_search": "web_search"})

Code thật (dùng đúng langgraph.StateGraph)

from typing import TypedDict, List, Annotated
import operator
from langgraph.graph import StateGraph, START, END


class BaoCaoState(TypedDict):
    quai_vat_thay: str
    bang_chung: str
    ban_nhap: str
    phe_binh: str
    so_lan_sua: int
    lich_su: Annotated[List[str], operator.add]


# Node 1: Reflexion-style tool call — lấy bằng chứng thật
def ngui_mui_tim_bang_chung(state: BaoCaoState) -> dict:
    ten = state["quai_vat_thay"]
    co_so_du_lieu_quai_vat = {
        "quy chan nhen": "co to nhen day nguy hiem, doc to lam te liet co",
        "quy trang nguyet": "chi manh khi co anh trang, so tieng chuong",
    }
    ket_qua = co_so_du_lieu_quai_vat.get(ten, "khong tim thay du lieu")
    return {"bang_chung": ket_qua, "lich_su": [f"[TOOL] -> {ket_qua}"]}


# Node 2: Generate — viết bản nháp
def viet_ban_nhap(state: BaoCaoState) -> dict:
    ban_nhap = f"Da gap {state['quai_vat_thay']}. Dac diem: {state['bang_chung']}."
    return {"ban_nhap": ban_nhap, "lich_su": [f"[GENERATE] {ban_nhap}"]}


# Node 3: bug hài hước — Inosuke chen ngang
def inosuke_pha_roi(state: BaoCaoState) -> dict:
    if state["so_lan_sua"] == 0:
        return {"ban_nhap": state["ban_nhap"] + " HA GUC 1 PHAT!!! (Inosuke gao)",
                "lich_su": ["[BUG] Inosuke chen cau vo nghia"]}
    return {"lich_su": ["[OK] Inosuke ngoi yen"]}


# Node 4: Reflect — Nezuko phê bình
def nezuko_phe_binh(state: BaoCaoState) -> dict:
    ban_nhap = state["ban_nhap"]
    if "gao" in ban_nhap.lower():
        phe_binh = "Thieu nghiem trong! Bo cau thua, ghi ro doc tinh."
    elif state["so_lan_sua"] < 1:
        phe_binh = "On, nhung can them muc do nguy hiem cu the."
    else:
        phe_binh = "DAT. San sang gui ve Truong Doan."
    return {"phe_binh": phe_binh, "so_lan_sua": state["so_lan_sua"]+1,
            "lich_su": [f"[REFLECT] {phe_binh}"]}


def can_sua_khong(state: BaoCaoState) -> str:
    if "DAT" in state["phe_binh"] or state["so_lan_sua"] >= 3:
        return "xong"
    return "sua_tiep"


graph = StateGraph(BaoCaoState)
graph.add_node("ngui_mui", ngui_mui_tim_bang_chung)
graph.add_node("viet_nhap", viet_ban_nhap)
graph.add_node("inosuke", inosuke_pha_roi)
graph.add_node("nezuko_cham", nezuko_phe_binh)
graph.add_edge(START, "ngui_mui")
graph.add_edge("ngui_mui", "viet_nhap")
graph.add_edge("viet_nhap", "inosuke")
graph.add_edge("inosuke", "nezuko_cham")
graph.add_conditional_edges("nezuko_cham", can_sua_khong,
    {"sua_tiep": "viet_nhap", "xong": END})

app = graph.compile()
ket_qua = app.invoke({"quai_vat_thay": "quy chan nhen", "bang_chung": "",
    "ban_nhap": "", "phe_binh": "", "so_lan_sua": 0, "lich_su": []})

Tại sao code này chạy được

LangGraph không quan tâm tên field là "bang_chung" hay "evidence", tên node là "ngui_mui" hay "tool_call" — nó chỉ quan tâm 3 thứ: (1) mỗi node là 1 hàm nhận state và trả về dict field cần cập nhật, (2) add_conditional_edges gọi đúng hàm route và tìm đúng key trong mapping, (3) reducer (operator.add) quyết định field được cộng dồn hay ghi đè. Vì 3 nguyên lý này được tuân thủ — dù đặt tên tiếng Việt theo theme Kimetsu no Yaiba — graph vẫn compile và chạy y hệt đặt tên tiếng Anh chuẩn. Đây không phải phép màu của "chủ đề hài hước", mà do nắm đúng cấu trúc dữ liệu và thứ tự thực thi — thứ LangGraph thực sự kiểm tra.
